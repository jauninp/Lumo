Add-Type -AssemblyName System.Windows.Forms

function Send-Key {
    param(
        [Parameter(Mandatory)]
        [string]$Key,

        [int]$Delay = 100
    )

    [System.Windows.Forms.SendKeys]::SendWait($Key)

    if ($Delay -gt 0) {
        Start-Sleep -Milliseconds $Delay
    }
}


function Send-Text {
    param(
        [AllowEmptyString()]
        [string]$Text,

        [int]$Delay = 100
    )

    if ($Text -ne "") {
        [System.Windows.Forms.SendKeys]::SendWait($Text)
    }

    if ($Delay -gt 0) {
        Start-Sleep -Milliseconds $Delay
    }
}


function Next-Field {
    param(
        [ValidateSet("Fast","Normal","Load")]
        [string]$Wait = "Fast"
    )

    Send-Key "{TAB}" 0

    switch ($Wait) {

        "Fast" {
            Start-Sleep -Milliseconds 150
        }

        "Normal" {
            Start-Sleep -Milliseconds 500
        }

        "Load" {
            # Temporaire :
            # plus tard, remplacer par une attente intelligente
            # (curseur Finnova occupe, changement de fenetre, etc.)
            Start-Sleep -Milliseconds 1500
        }
    }
}


function Write-TestLog {
    param(
        [string]$TestId,
        [string]$Message,
        [ValidateSet("INFO","OK","KO")]
        [string]$Status = "INFO"
    )

    $time = Get-Date -Format "HH:mm:ss"

    Write-Host "[$time] [$TestId] [$Status] $Message"
}

function Assert-FinnovaWindow {
    param(
        [Parameter(Mandatory)]
        [string]$Title
    )

    $p = Get-Process |
        Where-Object {
            $_.MainWindowTitle -like "*$Title*"
        } |
        Select-Object -First 1

    if (-not $p) {
        throw "Fenetre Finnova introuvable : $Title"
    }

    return $p
}


function Wait-FinnovaWindow {
    param(
        [Parameter(Mandatory)]
        [string]$Title,

        [int]$Timeout = 15
    )

    $end = (Get-Date).AddSeconds($Timeout)

    while ((Get-Date) -lt $end) {

        $p = Get-Process |
            Where-Object {
                $_.MainWindowTitle -like "*$Title*"
            } |
            Select-Object -First 1

        if ($p) {
            return $p
        }

        Start-Sleep -Milliseconds 200
    }

    throw "Timeout : fenetre '$Title' non trouvee"
}

function Fill-OrderCH {
    param($Test)

    Write-TestLog $Test.TestId "Saisie ordre CH"

    Send-Text $Test.Compte
    Next-Field Load

    Send-Text $Test.IBAN
    Next-Field Load

    Send-Text $Test.Montant
    Next-Field Fast

    Send-Text $Test.Devise
    Next-Field Normal

    Send-Text $Test.Date
    Next-Field Fast

    Send-Text $Test.Motif
}


function Validate-OrderCH {
    param($Test)

    Write-TestLog $Test.TestId "Validation ordre CH"

    Send-Key "{ENTER}"

    # Exemple :
    # adapter le titre de la popup Finnova
    Wait-FinnovaWindow "Confirmation" | Out-Null

    Send-Key "{ENTER}"

    Write-TestLog $Test.TestId "Ordre CH valide" "OK"
}


function Fill-OrderEUR {
    param($Test)

    Write-TestLog $Test.TestId "Saisie ordre EUR"

    Send-Text $Test.Compte
    Next-Field Load

    # Exemple : ordre des champs different du CH

    Send-Text $Test.Devise
    Next-Field Normal

    Send-Text $Test.IBAN
    Next-Field Load

    Send-Text $Test.Montant
    Next-Field Fast

    Send-Text $Test.Date
    Next-Field Fast

    Send-Text $Test.Motif
}


function Validate-OrderEUR {
    param($Test)

    Write-TestLog $Test.TestId "Validation ordre EUR"

    Send-Key "{ENTER}"

    # Exemple :
    # l'ordre EUR peut avoir une popup supplementaire.

    Wait-FinnovaWindow "Controle devise" | Out-Null
    Send-Key "{ENTER}"

    Wait-FinnovaWindow "Confirmation" | Out-Null
    Send-Key "{ENTER}"

    Write-TestLog $Test.TestId "Ordre EUR valide" "OK"
}


function Fill-OrderInterne {
    param($Test)

    Write-TestLog $Test.TestId "Saisie ordre interne"

    Send-Text $Test.Compte
    Next-Field Load

    Send-Text $Test.Montant
    Next-Field Fast

    Send-Text $Test.Motif
}


function Validate-OrderInterne {
    param($Test)

    Send-Key "{ENTER}"

    Wait-FinnovaWindow "Confirmation" | Out-Null
    Send-Key "{ENTER}"

    Write-TestLog $Test.TestId "Ordre interne valide" "OK"
}


function Invoke-Injection {
    param($Test)

    switch ($Test.Type.ToUpper()) {

        "CH" {
            Assert-FinnovaWindow "Ordre" | Out-Null

            Fill-OrderCH $Test
            Validate-OrderCH $Test
        }

        "EUR" {
            Assert-FinnovaWindow "Ordre" | Out-Null

            Fill-OrderEUR $Test
            Validate-OrderEUR $Test
        }

        "INTERNE" {
            Assert-FinnovaWindow "Ordre" | Out-Null

            Fill-OrderInterne $Test
            Validate-OrderInterne $Test
        }

        default {
            throw "Type inconnu : $($Test.Type)"
        }
    }
}


$tests = Import-Csv "$PSScriptRoot\inject-tests.csv" -Delimiter ";"

foreach ($test in $tests) {

    Write-Host ""
    Write-Host "====================================="
    Write-Host "TEST $($test.TestId) / $($test.Type)"
    Write-Host "====================================="

    try {

        Invoke-Injection $test

        Write-TestLog `
            $test.TestId `
            "Injection terminee" `
            "OK"
    }
    catch {

        Write-TestLog `
            $test.TestId `
            $_.Exception.Message `
            "KO"

        Write-Host ""
        Write-Host "ARRET DES TESTS"

        # Important :
        # si Finnova est dans un etat inattendu,
        # on ne lance pas automatiquement le test suivant.
        break
    }
}

============================================================
EVOLUTIONS A FAIRE ENSUITE
============================================================

1. NEXT-FIELD / ATTENTE INTELLIGENTE
------------------------------------

Pour l'instant :

    Next-Field Load

attend 1500 ms.

L'objectif sera de remplacer cela par une vraie attente Finnova :

    TAB
      |
      v
    Finnova charge
      |
      v
    curseur "occupe"
      |
      v
    attendre
      |
      v
    curseur normal
      |
      v
    continuer

On pourra garder un timeout maximum pour eviter de rester bloque.


2. LECTURE DU CHAMP COURANT
---------------------------

Ajouter une fonction du genre :

    Get-CurrentFieldValue

Elle pourra essayer :

    Ctrl+A
    Ctrl+C
    lecture du presse-papiers

Cela permettra de faire de vraies verifications :

    $montant = Get-CurrentFieldValue

    if ($montant -ne $Test.MontantAttendu) {
        throw "Montant incorrect"
    }


3. SEPARATION INJECTION / VERIFICATION
--------------------------------------

Les deux outils doivent rester independants.

Cas possibles :

A)
    injection automatique
    verification automatique

B)
    injection manuelle par un collegue
    verification automatique

C)
    injection automatique
    verification manuelle


4. IDENTIFIANT DE TEST
----------------------

Quand c'est possible, inserer une reference claire dans Finnova :

    AUTO-T001
    AUTO-T002
    AUTO-T003

Cela simplifie fortement la recherche dans la fenetre de controle.


5. REGLE DE SECURITE
--------------------

Si :
- mauvaise fenetre
- mauvais titre
- mauvais focus
- popup inattendue
- timeout
- verification incorrecte

ALORS :

    STOP

Ne jamais continuer automatiquement avec le test suivant si Finnova
n'est plus dans un etat connu.


6. TYPES D'ORDRE
----------------

Ne pas chercher tout de suite a faire un moteur trop abstrait.

Garder des fonctions explicites :

    Fill-OrderCH
    Validate-OrderCH

    Fill-OrderEUR
    Validate-OrderEUR

    Fill-OrderInterne
    Validate-OrderInterne

Cela sera plus facile a lire et a debugger.

Si plusieurs types deviennent tres similaires, on pourra factoriser ensuite.


7. LIBRAIRIE EXISTANTE
----------------------

Tu as deja une librairie qui controle notamment :
- le focus
- le nom de la fenetre
- la bonne fenetre
- etc.

Il faudra reutiliser cette librairie et remplacer progressivement
les fonctions d'exemple :

    Assert-FinnovaWindow
    Wait-FinnovaWindow

par tes fonctions existantes.
